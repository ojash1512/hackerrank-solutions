# Pitfalls    (1) Incorrect loop bounds, such as using range(6) instead of range(4), will cause an IndexError when accessing indices i+2 or j+2.  (2) Initializing max_sum to 0 instead of negative infinity will produce incorrect results when all hourglass sums are negative.

#!/bin/python3

import math
import os
import random
import re
import sys

#
# Complete the 'hourglassSum' function below.
#
# The function is expected to return an INTEGER.
# The function accepts 2D_INTEGER_ARRAY arr as parameter.
#

def hourglassSum(arr):
    # Write your code here
    max_sum = float('-inf')
    
    for i in range(4):
        for j in range(4):
            total = (
                arr[i][j] + arr[i][j+1] + arr[i][j+2]
                +arr[i+1][j+1]
                +arr[i+2][j] + arr[i+2][j+1] + arr[i+2][j+2]
            )
            max_sum = max(max_sum, total)
    return max_sum

if __name__ == '__main__':
    fptr = open(os.environ['OUTPUT_PATH'], 'w')

    arr = []

    for _ in range(6):
        arr.append(list(map(int, input().rstrip().split())))

    result = hourglassSum(arr)

    fptr.write(str(result) + '\n')

    fptr.close()
